static int CleanupStaleLogs(char const* path)
{
    ...
    for (auto const& item : entry)
    {
        if (!item.is_regular_file(ec)) { ec.clear(); continue; }   // 目录 / 子目录不碰
        if (item.path().extension() != ".log") { continue; }        // 只认 .log
        std::error_code removeError;
        if (std::filesystem::remove(item.path(), removeError) && !removeError) { ++removed; }
    }
    return removed;
}
⇒ 非 .log（json/png 等）与子目录在构造上不可能被删；删除失败只影响计数，不误删。
